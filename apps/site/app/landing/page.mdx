import { Form } from '@inertiajs/react'
import { href } from 'fondu/client'

export default function Todos({ items }: { items: Todo[] }) {
  return (
    <>
      <ul>
        {items.map((t) => (
          <li key={t.id}>{t.text}</li>
        ))}
      </ul>
      <Form action={href('/todos')} method="post" resetOnSuccess>
        <input name="text" />
        <button>Add</button>
      </Form>
    </>
  )
}
